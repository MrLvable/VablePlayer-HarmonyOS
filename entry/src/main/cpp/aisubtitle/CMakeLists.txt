cmake_minimum_required(VERSION 3.16)
project(ai_audio_extractor LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(AI_MPV_LIBRARY "" CACHE FILEPATH "Installed arm64 libmpv.so with FFmpeg n8.0 exports")
if(NOT AI_MPV_LIBRARY)
  get_filename_component(AI_ENTRY_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
  set(AI_MPV_LIBRARY "${AI_ENTRY_DIR}/oh_modules/@mpv-ohos/mpv-arkts/libs/arm64-v8a/libmpv.so")
endif()
if(NOT EXISTS "${AI_MPV_LIBRARY}")
  message(FATAL_ERROR "Install @mpv-ohos/mpv-arkts or set AI_MPV_LIBRARY to its arm64 libmpv.so")
endif()
add_library(ai_mpv_import SHARED IMPORTED)
set_target_properties(ai_mpv_import PROPERTIES IMPORTED_LOCATION "${AI_MPV_LIBRARY}")
add_library(ai_audio_extractor SHARED audio_extractor.cpp)
target_include_directories(ai_audio_extractor PRIVATE "${CMAKE_CURRENT_LIST_DIR}/third_party/ffmpeg/include")
target_compile_options(ai_audio_extractor PRIVATE -Wall -Wextra -Werror=return-type)
target_link_libraries(ai_audio_extractor PRIVATE ai_mpv_import ace_napi.z)
